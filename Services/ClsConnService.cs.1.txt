        private async Task StartBackgroundReconnectTaskAsync()
        {
            if (!await mReconnectLock.WaitAsync(0))
            {
                return;
            }

            try
            {
                if (Connectivity.Current.NetworkAccess == NetworkAccess.Internet)
                {
                    bool isReconnected = await MauiRedirectAsync();
                    if (isReconnected)
                    {
                        mServerError = false;
                    }
                }
            }
            catch (Exception ex)
            {
                // [Ghi chú]: Do đã chuyển sang AsyncLocal nên gán LastMessage ở đây an toàn tuyệt đối
                LastMessage = $"Background Reconnect Exception: {ex.Message}";
                LogWriter.WriteLine(LastMessage);
            }
            finally
            {
                mReconnectLock.Release();
            }
        }

        /// <summary>
        /// Hàm POST chính dùng cho cả Tuần tự (Login) và Ngầm (Sự kiện)
        /// TẤT CẢ các biến trong hàm này (mUrl, client, response, result) đều là
        /// biến CỤC BỘ (Stack Variable), hoàn toàn độc lập 100% giữa các Task.
        /// </summary>
        public async Task<string> SubmitMauiAsync(string mTrueUrlService, string jsonPost)
        {
            // Reset LastMessage riêng của Task/Luồng này
            LastMessage = "";

            if (string.IsNullOrEmpty(jsonPost))
            {
                LastMessage = "SubmitMaui: JSON POST rỗng";
                return "";
            }
            if (string.IsNullOrEmpty(mTrueUrlService))
            {
                LastMessage = "SubmitMaui: URL rỗng";
                return "";
            }

            if (Connectivity.Current.NetworkAccess != NetworkAccess.Internet)
            {
                mNetError = true;
                LastMessage = "SubmitMaui: Mất kết nối Internet";
                return "";
            }

            // [AN TOÀN ĐA LUỒNG]: mUrl là biến CỤC BỘ -> Không sợ bị đè biến URL
            int nReaden = mTrueUrlService.IndexOf('?');
            if (nReaden >= 0)
            {
                mTrueUrlService = mTrueUrlService.Substring(0, nReaden);
            }
            mTrueUrlService = mTrueUrlService.TrimEnd('/');
            string mUrl = mTrueUrlService + "/SubmitMaui";

            const int maxRetries = 2;
            const int perRequestTimeoutSeconds = 2;

            for (int attempt = 1; attempt <= maxRetries; attempt++)
            {
                try
                {
                    // [AN TOÀN ĐA LUỒNG]: Mỗi Task tạo 1 HttpClient Handler độc lập
                    using (var handler = new HttpClientHandler())
                    {
                        handler.AutomaticDecompression = System.Net.DecompressionMethods.GZip | System.Net.DecompressionMethods.Deflate;
                        handler.AllowAutoRedirect = true;
                        handler.UseCookies = false;

                        using (var client = new HttpClient(handler))
                        {
                            client.Timeout = TimeSpan.FromSeconds(perRequestTimeoutSeconds);
                            string userAgent = !string.IsNullOrWhiteSpace(mSoapUserAgent) ? mSoapUserAgent : "BI.Application";
                            client.DefaultRequestHeaders.UserAgent.ParseAdd(userAgent);

                            using (var content = new StringContent(jsonPost, Encoding.UTF8, "application/json"))
                            {
                                using (var response = await client.PostAsync(mUrl, content))
                                {
                                    int statusCode = (int)response.StatusCode;

                                    if (statusCode >= 500 && statusCode < 600)
                                    {
                                        mServerError = true;
                                        _ = StartBackgroundReconnectTaskAsync();

                                        LastMessage = $"SubmitMaui Server Error Status: {statusCode} {response.ReasonPhrase}";
                                        if (attempt < maxRetries) continue;
                                        return "";
                                    }

                                    if (!response.IsSuccessStatusCode)
                                    {
                                        string errorBody = "";
                                        try { errorBody = await response.Content.ReadAsStringAsync(); } catch { }

                                        LastMessage = $"SubmitMaui HTTP Error: {statusCode} {response.ReasonPhrase}";
                                        if (!string.IsNullOrEmpty(errorBody))
                                        {
                                            LastMessage += " | Response=" + errorBody;
                                        }
                                        return "";
                                    }

                                    string result = await response.Content.ReadAsStringAsync();
                                    if (string.IsNullOrEmpty(result))
                                    {
                                        LastMessage = "SubmitMaui: Server trả body rỗng";
                                        return "";
                                    }

                                    mServerError = false;
                                    LastMessage = "";
                                    return result; // [AN TOÀN ĐA LUỒNG]: Trả về kết quả đúng của Task này
                                }
                            }
                        }
                    }
                }
                catch (TaskCanceledException ex)
                {
                    LastMessage = $"SubmitMaui Attempt {attempt} Timeout (>{perRequestTimeoutSeconds}s): {ex.Message}";
                    LogWriter.WriteLine(LastMessage);

                    if (attempt == maxRetries)
                    {
                        mServerError = true;
                        _ = StartBackgroundReconnectTaskAsync();
                    }
                }
                catch (HttpRequestException ex)
                {
                    LastMessage = $"SubmitMaui Attempt {attempt} HttpRequestException: {ex.Message}";
                    LogWriter.WriteLine(LastMessage);

                    mServerError = true;
                    _ = StartBackgroundReconnectTaskAsync();

                    if (attempt == maxRetries) return "";
                }
                catch (Exception ex)
                {
                    if (ex.InnerException is SocketException)
                    {
                        mServerError = true;
                        _ = StartBackgroundReconnectTaskAsync();
                    }

                    LastMessage = $"SubmitMaui Exception: {ex.Message}";
                    LogWriter.WriteLine(LastMessage);
                    if (attempt == maxRetries) return "";
                }
            }

            return "";
        }

        private async Task<bool> UrlAvailableCheckAsync(string url, int timeoutInSecond = 10)
        {
            if (string.IsNullOrWhiteSpace(url)) return false;
            if (timeoutInSecond <= 0) timeoutInSecond = 10;

            return await Task.Run(async () =>
            {
                try
                {
                    using (var handler = new HttpClientHandler())
                    {
                        handler.ServerCertificateCustomValidationCallback = (sender, cert, chain, sslPolicyErrors) => true;
                        using (var client = new HttpClient(handler))
                        {
                            client.Timeout = TimeSpan.FromSeconds(timeoutInSecond);

                            string userAgent = !string.IsNullOrWhiteSpace(mSoapUserAgent) ? mSoapUserAgent : "BI.Application";
                            client.DefaultRequestHeaders.UserAgent.ParseAdd(userAgent);

                            using (var response = await client.GetAsync(url, HttpCompletionOption.ResponseHeadersRead).ConfigureAwait(false))
                            {
                                int status = (int)response.StatusCode;
                                return status >= 100 && status < 400;
                            }
                        }
                    }
                }
                catch (Exception ex)
                {
                    LastMessage = $"UrlAvailableCheck Exception: {ex.Message}";
                    LogWriter.WriteLine(LastMessage);
                    return false;
                }
            }).ConfigureAwait(false);
        }

        public async Task<bool> MauiRedirectAsync()
        {
            string availableRedirectUrl = await RedirectAvailableGetAsync();
            if (string.IsNullOrEmpty(availableRedirectUrl)) return false;

            DataSet ds = new DataSet();
            DataTable dt = new DataTable("Table1");
            dt.Columns.Add("RedirectCode", typeof(string));
            dt.Columns.Add("UserName", typeof(string));
            dt.Columns.Add("UserPass", typeof(string));

            DataRow dr = dt.NewRow();
            dr["RedirectCode"] = mRedirectCode;
            dr["UserName"] = mRedirectUserName;
            dr["UserPass"] = mRedirectUserPass;
            dt.Rows.Add(dr);
            ds.Tables.Add(dt);

            string jsonPost = ClienJsonPOST("", "MauiRedirect", ds);
            string jsonResponse = await SubmitMauiAsync(availableRedirectUrl, jsonPost);

            if (string.IsNullOrEmpty(jsonResponse)) return false;

            DataSet dsResult = ClienJsonParse(jsonResponse);
            if (dsResult != null && dsResult.Tables.Count > 0 && dsResult.Tables[0].Rows.Count > 0)
            {
                DataRow row = dsResult.Tables[0].Rows[0];
                if (row.Table.Columns.Contains("UrlTrueService"))
                    mUrlTrueService = row["UrlTrueService"].ToString();
                if (row.Table.Columns.Contains("SecurityCode"))
                    mSecurityCode = row["SecurityCode"].ToString();

                mServerError = false;
                return true;
            }

            return false;
        }

        public string ClienJsonPOST(string securityCode, string functionName)
        {
            return ClienJsonPOST(securityCode, functionName, null);

        public string ClienJsonPOST(string securityCode, string functionName, DataSet dsSet)
        {
            try
            {
                JObject root = new JObject
                {
                    { "SecurityCode", securityCode },
                    { "FunctionName", functionName }
                };

                if (dsSet != null && dsSet.Tables.Count > 0)
                {
                    JObject dataObj = new JObject();
                    int tableIndex = 1;

                    foreach (DataTable table in dsSet.Tables)
                    {
                        JArray rowsArray = new JArray();

                        foreach (DataRow row in table.Rows)
                        {
                            JObject rowObj = new JObject();
                            foreach (DataColumn col in table.Columns)
                            {
                                object val = row[col];
                                string strVal = string.Empty;

                                if (val != DBNull.Value && val != null)
                                {
                                    if (val is DateTime dateTimeVal)
                                    {
                                        strVal = dateTimeVal.ToString("yyyy-MM-dd HH:mm:ss");
                                    }
                                    else
                                    {
                                        strVal = val.ToString();
                                    }
                                }
                                rowObj.Add(col.ColumnName, strVal);
                            }
                            rowsArray.Add(rowObj);
                        }

                        string tableName = table.TableName;
                        if (string.IsNullOrEmpty(tableName))
                        {
                            tableName = "Table" + tableIndex.ToString();
                            tableIndex++;
                        }
                        dataObj.Add(tableName, rowsArray);
                    }
                    root.Add("data", dataObj);
                }
                else
                {
                    root.Add("data", JValue.CreateNull());
                }

                return root.ToString(Formatting.None);
            }
            catch (Exception ex)
            {
                LastMessage = ex.ToString();
                return "";
            }
        }

        public DataSet ClienJsonParse(string jsonResponse)
        {
            if (string.IsNullOrWhiteSpace(jsonResponse)) return null;

            try
            {
                DataSet ds = new DataSet();
                JObject root = JObject.Parse(jsonResponse);

                if (root["data"] != null && root["data"].Type == JTokenType.Object)
                {
                    JObject dataObj = (JObject)root["data"];
                    foreach (var property in dataObj.Properties())
                    {
                        DataTable dt = new DataTable(property.Name);
                        if (property.Value is JArray rowsArray && rowsArray.Count > 0)
                        {
                            JObject firstRow = (JObject)rowsArray[0];
                            foreach (var col in firstRow.Properties())
                            {
                                dt.Columns.Add(col.Name, typeof(string));
                            }

                            foreach (JObject rowObj in rowsArray)
                            {
                                DataRow dr = dt.NewRow();
                                foreach (var col in rowObj.Properties())
                                {
                                    dr[col.Name] = col.Value?.ToString() ?? "";
                                }
                                dt.Rows.Add(dr);
                            }
                        }
                        ds.Tables.Add(dt);
                    }
                }
                return ds;
            }
            catch (Exception ex)
            {
                LastMessage = "ClienJsonParse Error: " + ex.Message;
                return null;
            }
        }

                                        return "";
                                    }

                                    if (!response.IsSuccessStatusCode)
                                    {
                                        string errorBody = "";
                                        try { errorBody = await response.Content.ReadAsStringAsync(); } catch { }

                                        LastMessage = $"SubmitMaui HTTP Error: {statusCode} {response.ReasonPhrase}";
                                        if (!string.IsNullOrEmpty(errorBody))
                                        {
                                            LastMessage += " | Response=" + errorBody;
                                        }
                                        return "";
                                    }

                                    string result = await response.Content.ReadAsStringAsync();
                                    if (string.IsNullOrEmpty(result))
                                    {
                                        LastMessage = "SubmitMaui: Server trả body rỗng";
                                        return "";
                                    }

                                    mServerError = false;
                                    LastMessage = "";
                                    return result;
                                }
                            }
                        }
                    }
                }
                catch (TaskCanceledException ex)
                {
                    LastMessage = $"SubmitMaui Attempt {attempt} Timeout (>{perRequestTimeoutSeconds}s): {ex.Message}";
                    LogWriter.WriteLine(LastMessage);

                    if (attempt == maxRetries)
                    {
                        mServerError = true;
                        _ = StartBackgroundReconnectTaskAsync();
                    }
                }
                catch (HttpRequestException ex)
                {
                    LastMessage = $"SubmitMaui Attempt {attempt} HttpRequestException: {ex.Message}";
                    LogWriter.WriteLine(LastMessage);

                    mServerError = true;
                    _ = StartBackgroundReconnectTaskAsync();

                    if (attempt == maxRetries) return "";
                }
                catch (Exception ex)
                {
                    if (ex.InnerException is SocketException)
                    {
                        mServerError = true;
                        _ = StartBackgroundReconnectTaskAsync();
                    }

                    LastMessage = $"SubmitMaui Exception: {ex.Message}";
                    LogWriter.WriteLine(LastMessage);
                    if (attempt == maxRetries) return "";
                }
            }

            return "";
        }

        private async Task<bool> UrlAvailableCheckAsync(string url, int timeoutInSecond = 10)
        {
            if (string.IsNullOrWhiteSpace(url)) return false;
            if (timeoutInSecond <= 0) timeoutInSecond = 10;

            return await Task.Run(async () =>
            {
                try
                {
                    using (var handler = new HttpClientHandler())
                    {
                        handler.ServerCertificateCustomValidationCallback = (sender, cert, chain, sslPolicyErrors) => true;
                        using (var client = new HttpClient(handler))
                        {
                            client.Timeout = TimeSpan.FromSeconds(timeoutInSecond);

                            string userAgent = !string.IsNullOrWhiteSpace(mSoapUserAgent) ? mSoapUserAgent : "BI.Application";
                            client.DefaultRequestHeaders.UserAgent.ParseAdd(userAgent);

                            using (var response = await client.GetAsync(url, HttpCompletionOption.ResponseHeadersRead).ConfigureAwait(false))
                            {
                                int status = (int)response.StatusCode;
                                return status >= 100 && status < 400;
                            }
                        }
                    }
                }
                catch (Exception ex)
                {
                    LastMessage = $"UrlAvailableCheck Exception: {ex.Message}";
                    LogWriter.WriteLine(LastMessage);
                    return false;
                }
            }).ConfigureAwait(false);
        }

        public async Task<bool> MauiRedirectAsync()
        {
            string availableRedirectUrl = await RedirectAvailableGetAsync();
            if (string.IsNullOrEmpty(availableRedirectUrl)) return false;

            DataSet ds = new DataSet();
            DataTable dt = new DataTable("Table1");
            dt.Columns.Add("RedirectCode", typeof(string));
            dt.Columns.Add("UserName", typeof(string));
            dt.Columns.Add("UserPass", typeof(string));

            DataRow dr = dt.NewRow();
            dr["RedirectCode"] = mRedirectCode;
            dr["UserName"] = mRedirectUserName;
            dr["UserPass"] = mRedirectUserPass;
            dt.Rows.Add(dr);
            ds.Tables.Add(dt);

            string jsonPost = ClienJsonPOST("", "MauiRedirect", ds);
            string jsonResponse = await SubmitMauiAsync(availableRedirectUrl, jsonPost);

            if (string.IsNullOrEmpty(jsonResponse)) return false;

            DataSet dsResult = ClienJsonParse(jsonResponse);
            if (dsResult != null && dsResult.Tables.Count > 0 && dsResult.Tables[0].Rows.Count > 0)
            {
                DataRow row = dsResult.Tables[0].Rows[0];
                if (row.Table.Columns.Contains("UrlTrueService"))
                    mUrlTrueService = row["UrlTrueService"].ToString();
                if (row.Table.Columns.Contains("SecurityCode"))
                    mSecurityCode = row["SecurityCode"].ToString();

                mServerError = false;
                return true;
            }

            return false;
        }

        public string ClienJsonPOST(string securityCode, string functionName)
        {
            return ClienJsonPOST(securityCode, functionName, null);
        }

        public string ClienJsonPOST(string securityCode, string functionName, DataSet dsSet)
        {
            try
            {
                JObject root = new JObject
                {
                    { "SecurityCode", securityCode },
                    { "FunctionName", functionName }
                };

                if (dsSet != null && dsSet.Tables.Count > 0)
                {
                    JObject dataObj = new JObject();
                    int tableIndex = 1;

                    foreach (DataTable table in dsSet.Tables)
                    {
                        JArray rowsArray = new JArray();

                        foreach (DataRow row in table.Rows)
                        {
                            JObject rowObj = new JObject();
                            foreach (DataColumn col in table.Columns)
                            {
                                object val = row[col];
                                string strVal = string.Empty;

                                if (val != DBNull.Value && val != null)
                                {
                                    if (val is DateTime dateTimeVal)
                                    {
                                        strVal = dateTimeVal.ToString("yyyy-MM-dd HH:mm:ss");
                                    }
                                    else
                                    {
                                        strVal = val.ToString();
                                    }
                                }
                                rowObj.Add(col.ColumnName, strVal);
                            }
                            rowsArray.Add(rowObj);
                        }

                        string tableName = table.TableName;
                        if (string.IsNullOrEmpty(tableName))
                        {
                            tableName = "Table" + tableIndex.ToString();
                            tableIndex++;
                        }
                        dataObj.Add(tableName, rowsArray);
                    }
                    root.Add("data", dataObj);
                }
                else
                {
                    root.Add("data", JValue.CreateNull());
                }

                return root.ToString(Formatting.None);
            }
            catch (Exception ex)
            {
                LastMessage = ex.ToString();
                return "";
            }
        }

        public DataSet ClienJsonParse(string jsonResponse)
        {
            if (string.IsNullOrWhiteSpace(jsonResponse)) return null;

            try
            {
                DataSet ds = new DataSet();
                JObject root = JObject.Parse(jsonResponse);

                if (root["data"] != null && root["data"].Type == JTokenType.Object)
                {
                    JObject dataObj = (JObject)root["data"];
                    foreach (var property in dataObj.Properties())
                    {
                        DataTable dt = new DataTable(property.Name);
                        if (property.Value is JArray rowsArray && rowsArray.Count > 0)
                        {
                            JObject firstRow = (JObject)rowsArray[0];
                            foreach (var col in firstRow.Properties())
                            {
                                dt.Columns.Add(col.Name, typeof(string));
                            }

                            foreach (JObject rowObj in rowsArray)
                            {
                                DataRow dr = dt.NewRow();
                                foreach (var col in rowObj.Properties())
                                {
                                    dr[col.Name] = col.Value?.ToString() ?? "";
                                }
                                dt.Rows.Add(dr);
                            }
                        }
                        ds.Tables.Add(dt);
                    }
                }
                return ds;
            }
            catch (Exception ex)
            {
                LastMessage = "ClienJsonParse Error: " + ex.Message;
                return null;
            }
        }

        public async Task<bool> LoadByWakeup()
        {
            if (string.IsNullOrEmpty(mSecurityCode))

                mSecurityCode = AppSettings.SecurityCode;

                    return false;
                }
                return true;
            }
            return false;
        }

            mRedirectUrlAvailable2 = AppSettings.RedirectUrlAvailable2;

                mUserPass = AppSettings.LoginUserPass;
            }
            else
            {
                mUserName = AppSettings.LoginUserName;
